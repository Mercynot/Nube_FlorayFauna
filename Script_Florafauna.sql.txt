
CREATE DATABASE ecologia_chile;
USE ecologia_chile;

CREATE TABLE especies (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre_comun VARCHAR(100),
    reino ENUM('Flora', 'Fauna'),
    nombre_cientifico VARCHAR(100),
    estado_conservacion VARCHAR(50)
);
